# Limits the schema cannot express, for pbview's typed encoders (pbview_library(bounds = ...)):
# `<package>.<Message>.<field> max_len N` (bytes / string) and `max_count N` (repeated). The
# setters enforce them and a fully bounded message gets FooEnc::MAX_ENCODED_LEN. The numbers
# are the budgets of firmware/mesh/swarm/docs/swarm-wire.md.

# LinkFrame (upstream: C++ only; Splanc: also firmware/mesh/src/link.rs). Splanc raised the
# payload from the UWB frame's 115 to a MeshPacket at the largest capacity, 25 + 1400
# (mesh.rs PAYLOAD_CAP, design §5.2); each backhaul's MTU is the real limit.
swarm.v1.LinkFrame.payload max_len 1425
swarm.v1.LinkFrame.advert max_len 64
swarm.v1.LinkFrame.summary max_len 26

# Firmware distribution
swarm.v1.ManifestBody.sha256 max_len 32
swarm.v1.Manifest.body max_len 46

swarm.v1.FwData.data max_len 96

# Mesh. Splanc raised the payload from the swarm's 88 (UWB) to mesh.rs PAYLOAD_CAP: an
# instance's capacity and its backhaul's MTU are the real limits (budget.rs asserts both).
swarm.v1.MeshPacket.payload max_len 1400
