# Code (and its rodata) the prebuilt SDK's sdkconfig places in internal RAM that
# this firmware never runs with the flash cache disabled. Each move is the
# linker-script effect of an IDF Kconfig placement option we can't turn on the
# prebuilt libs. Evidence: the IRAM audit (docs/design/ram-budget.md) — none of
# these is reachable from an interrupt that is registered IRAM-safe in the
# min-watermark RAM dump, nor from the flash-write path.
#
# WiFi power-save / sleep paths (.wifislpiram: the TX/timer half of
# CONFIG_ESP_WIFI_SLP_IRAM_OPT=n): the netstack runs WIFI_PS_NONE with the radio
# forced awake. The .wifislprxiram half stays — it also carries the vendor's
# per-frame RX processing (wDev_ProcessRxSucData, ppRxPkt, ...).
libpp.a .wifislpiram
libnet80211.a .wifislpiram
# The power-save state machine itself (TBTT / beacon-miss / TWT bookkeeping)
# shares the library-wide .wifislprxiram / .wifiextrairam placement with live
# per-frame code, so it is carved out per object.
libpp.a:pm .wifislprxiram .wifiextrairam
libpp.a:pm_twt .wifislprxiram .wifiextrairam
# CONFIG_RINGBUF_PLACE_FUNCTIONS_INTO_FLASH=y: the only ring buffer user is the
# USB-serial console driver, whose ISR is not IRAM-safe.
libesp_ringbuf.a
# CONFIG_RMT_ISR_IRAM_SAFE=n: the LED output is PARLIO on the C6; RMT is only
# linked via Arduino's RGB_BUILTIN helper (the ws2812_rmt_forced debug build
# runs RMT without an IRAM-safe ISR, i.e. its LED ISR waits out flash writes).
libesp_driver_rmt.a:rmt_tx
libesp_driver_rmt.a:rmt_rx
libesp_driver_rmt.a:rmt_encoder
# CONFIG_I2C_MASTER_ISR_HANDLER_IN_IRAM=y but CONFIG_I2C_ISR_IRAM_SAFE=n: the
# sensor core's I2C interrupt (sensor_bus.cpp) is allocated WITHOUT
# ESP_INTR_FLAG_IRAM, so it is masked while the flash cache is off and its
# handler never needs IRAM (724 B back; docs/design/sensor-core.md budget).
libesp_driver_i2c.a:i2c_master
# Flash on the general-purpose SPI host: we only use the main (SPI1) flash.
# (The other flash VENDORS' chip drivers stay: a board fitted with, say, a
# Winbond part runs that driver with the cache off on every flash write.)
libhal.a:spi_flash_hal_gpspi

# --- the other direction: our own hot code pinned INTO IRAM -------------------
# The per-LED fx shade entry (single-deck path: lm_fx_shade with the deck shade
# inlined) plus the two fx_vm calls it makes per LED before the interpreter.
# Flash-cache misses re-warming this path every frame cost fx_bench ~15% on
# short sweeps (sweep16) and moved with unrelated layout changes; ~2.2 KB of
# IRAM makes it layout-independent. The interpreter (`run`, 37 KB) stays in
# flash, as do the two-deck blend and every cold path.
iram: .text.lm_fx_shade .text._ZN15ledmapper_fx_vm7Program5parse* .text._ZN15ledmapper_fx_vm2Vm17run_shade_bounded*

# --- the per-frame render path that stays in flash, pinned FIRST in .flash.text ---
# Everything the active-effect frame runs from flash outside the IRAM shade entry:
# render_once + cc_apply (main.cpp), the player FFI's per-frame calls and the fx_vm
# interpreter. Left to link order these sat after all of main.cpp and
# netstack_transport.cpp, so ANY growth there shifted them and changed how many
# flash-cache lines each frame re-warms: #311's +~4 KB of app code moved fx_bench
# by a constant ~+9 k cycles per frame (sweep16 +16%, past its 15% margin) and a
# pure-padding build of the old code reproduced it exactly. At the start of
# .flash.text their addresses no longer depend on the app code in front of them.
# The interpreter itself is a free function (ledmapper_fx_vm::run, mangled
# _ZN15ledmapper_fx_vm3run…), as are load_ctx, the memo ops and the math helpers,
# so the `2Vm*` (Vm methods) glob alone left ~40 KB of the hot path at the END of
# the Rust player: #366's ~1.8 KB of new app code shifted `run` and moved fx_bench's
# sweep16 by ~+6 k cycles (+18 %) with no change on the frame path. `fx_vm[0-9]*`
# pins the whole fx_vm crate (every item path starts with a length digit).
flash-first: .text._ZL11render_oncev .text._ZL8cc_applyPKh .text.lm_fx_update .text.lm_fx_last_update_outcome .text.lm_map_len .text.lm_map_led .text.lm_time_base .text._ZN20ledmapper_player_ffi16fx_advance_morph* .text._ZN15ledmapper_fx_vm2Vm* .text._ZN15ledmapper_fx_vm[0-9]*
